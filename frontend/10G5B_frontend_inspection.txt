import type {
  FeederCalibration,
  FeederCalibrationInput,
  FeederCalibrationUpdateInput,
  FeedingEvent,
  FeedingSchedule,
  FeedingScheduleInput,
  FeedingScheduleUpdateInput,
} from '../types/feeding'

import {
  apiRequest,
} from './apiClient'


export const FEEDING_HISTORY_PAGE_SIZE = 50

export const CALIBRATION_HISTORY_PAGE_SIZE = 50


export interface FeedingEventHistoryPage {
  events: FeedingEvent[]
  hasNext: boolean
  hasPrevious: boolean
  offset: number
}


export interface FeederCalibrationHistoryPage {
  calibrations: FeederCalibration[]
  hasNext: boolean
  hasPrevious: boolean
  offset: number
}


export async function listFeedingSchedules(
  systemId: string,
): Promise<FeedingSchedule[]> {
  const encodedSystemId =
    encodeURIComponent(
      systemId,
    )

  return apiRequest<FeedingSchedule[]>(
    `feeding/schedules?system_id=${encodedSystemId}`,
  )
}


export async function createFeedingSchedule(
  systemId: string,
  input: FeedingScheduleInput,
): Promise<FeedingSchedule> {
  return apiRequest<FeedingSchedule>(
    'feeding/schedules',
    {
      method: 'POST',

      body: JSON.stringify({
        system_id: systemId,
        ...input,
      }),
    },
  )
}


export async function updateFeedingSchedule(
  scheduleId: string,
  input: FeedingScheduleUpdateInput,
): Promise<FeedingSchedule> {
  return apiRequest<FeedingSchedule>(
    `feeding/schedules/${encodeURIComponent(scheduleId)}`,
    {
      method: 'PATCH',
      body: JSON.stringify(input),
    },
  )
}


export async function deleteFeedingSchedule(
  scheduleId: string,
): Promise<void> {
  await apiRequest<void>(
    `feeding/schedules/${encodeURIComponent(scheduleId)}`,
    {
      method: 'DELETE',
    },
  )
}


export async function loadFeederCalibrationHistoryPage(
  actuatorId: string,
  offset: number,
): Promise<FeederCalibrationHistoryPage> {
  const normalizedOffset =
    Math.max(
      0,
      Math.trunc(offset),
    )

  const encodedActuatorId =
    encodeURIComponent(
      actuatorId,
    )

  const calibrations =
    await apiRequest<FeederCalibration[]>(
      `feeding/calibrations/history?actuator_id=${encodedActuatorId}` +
      `&limit=${CALIBRATION_HISTORY_PAGE_SIZE + 1}` +
      `&offset=${normalizedOffset}`,
    )

  return {
    calibrations:
      calibrations.slice(
        0,
        CALIBRATION_HISTORY_PAGE_SIZE,
      ),

    hasNext:
      calibrations.length >
      CALIBRATION_HISTORY_PAGE_SIZE,

    hasPrevious:
      normalizedOffset > 0,

    offset:
      normalizedOffset,
  }
}

   * Load systems once.

  useEffect(() => {
    let active = true

    async function load() {
      try {
        const result =
          await listSystems()

        if (!active) {
          return
        }

        const ordered =
          [...result].sort(
            (left, right) =>
              left.name.localeCompare(
                right.name,

          )

        setSystems(
          ordered,
        )

        setSelection({
          systemId:
            ordered[0]?.id ?? null,

          offset: 0,
          revision: 0,
        })

        setSystemsStatus(
          'ready',
        )
      } catch {
        if (active) {
          setSystemsStatus(
            'error',
          )
        }
      }
    }

    void load()

    return () => {
      active = false
    }
  }, [])


  /*
   * Load all configured, non-deleted actuators
   * belonging to the selected system.
   *
   * Do not guess which actuator types are valid
   * feeders. The backend relationship contract
   * remains authoritative.
   */
  useEffect(() => {
    const systemId =
      selection.systemId

    if (systemId === null) {
      return
    }

    const selectedSystemId =
      systemId

    let active = true

    async function load() {
      try {
        const result =
          await loadSystemActuatorOverview(
            selectedSystemId,
          )

        if (!active) {
          return
        }

        const orderedActuators =
          [...result.actuators].sort(
            (left, right) =>
              left.name.localeCompare(
                right.name,
              ),
          )

        setActuatorState({
          systemId:
            selectedSystemId,

          status: 'ready',

          actuators:
            orderedActuators,
        })

        setCalibrationSelection({
          actuatorId: null,
          offset: 0,
        })

        setCalibrationSelection(
          (current) => {
            const actuatorStillAvailable =
              orderedActuators.some(
                (actuator) =>
                  actuator.id ===
                  current.actuatorId,
              )
            if (actuatorStillAvailable) {
              return current
            }

            return {
              actuatorId:
                orderedActuators[0]?.id ??
                null,

              offset: 0,
            }
          },
        )

      } catch {
        if (!active) {
          return
        }

        setActuatorState({
          systemId:
            selectedSystemId,

          status: 'error',
          actuators: [],
        })
      }
    }

    void load()

    return () => {
      active = false
    }
  }, [
    selection.systemId,

      setMutationState(
        null,

    } catch (error) {
      setMutationState({
        systemId,
        status: 'error',

        errorMessage:
          'Unable to create feeding schedule.',

      throw error
    }
  }

  async function createCalibration(
  input: FeederCalibrationInput,
): Promise<FeederCalibration> {
  const systemId =
    selection.systemId

  const actuatorId =
    calibrationSelection.actuatorId

  if (
    systemId === null ||
    actuatorId === null
  ) {
    throw new Error(
      'No actuator selected for calibration.',
    )
  }

  setMutationState({
    systemId,
    status: 'saving',
    errorMessage: null,
  })

  try {
    const calibration =
      await createFeederCalibration(
        actuatorId,
        input,
      )

    setCalibrationRevision(
      (current) =>
        current + 1,
    )

    setMutationState(
      null,
    )

    return calibration
  } catch (error) {
    setMutationState({
      systemId,
      status: 'error',

      errorMessage:
        'Unable to create feeder calibration.',
    })

    throw error
  }
}


async function updateCalibration(
  calibrationId: string,
  input: FeederCalibrationUpdateInput,
): Promise<FeederCalibration> {
  const systemId =
    selection.systemId

  if (systemId === null) {
    throw new Error(
      'No aquaponics system selected.',
    )
  }

  setMutationState({
    systemId,
    status: 'saving',
    errorMessage: null,
  })

  try {
    const calibration =
      await updateFeederCalibration(
        calibrationId,
        input,
      )

    setCalibrationRevision(
      (current) =>
        current + 1,
    )

    setMutationState(
      null,
    )

    return calibration
  } catch (error) {
    setMutationState({
      systemId,
      status: 'error',

      errorMessage:
        'Unable to update feeder calibration.',
    })

    throw error
  }
}


async function deleteCalibration(
  calibrationId: string,
): Promise<void> {
  const systemId =
    selection.systemId

  if (systemId === null) {
    throw new Error(
      'No aquaponics system selected.',
    )
  }

  setMutationState({
    systemId,
    status: 'deleting',
    errorMessage: null,
  })

  try {
    await deleteFeederCalibration(
      calibrationId,
    )

    setCalibrationRevision(
      (current) =>
        current + 1,
    )

    setMutationState(
      null,
    )
  } catch (error) {
    setMutationState({
      systemId,
      status: 'error',

      errorMessage:
        'Unable to delete feeder calibration.',
    })

    throw error
  }
}

function formatTimestamp(
  value: string,
): string {
  const date =
    new Date(value)

  return Number.isNaN(
    date.getTime(),
  )
    ? value
    : date.toLocaleString()
}


function formatNullableValue(
  value:
    number |
    string |
    null,
): string {
  return value === null
    ? 'Not recorded'
    : String(value)
}
