import type {
  FeedingEvent,
  FeedingSchedule,
  FeedingScheduleInput,
  FeedingScheduleUpdateInput,
} from '../types/feeding'

import {
  apiRequest,
} from './apiClient'


export const FEEDING_HISTORY_PAGE_SIZE = 50


export interface FeedingEventHistoryPage {
  events: FeedingEvent[]
  hasNext: boolean
  hasPrevious: boolean
  offset: number
}


export async function listFeedingSchedules(
  systemId: string,
): Promise<FeedingSchedule[]> {
  const encodedSystemId =
    encodeURIComponent(
      systemId,
    )

  return apiRequest<FeedingSchedule[]>(
    `feeding/schedules?system_id=${encodedSystemId}`,
  )
}


export async function createFeedingSchedule(
  systemId: string,
  input: FeedingScheduleInput,
): Promise<FeedingSchedule> {
  return apiRequest<FeedingSchedule>(
    'feeding/schedules',
    {
      method: 'POST',

      body: JSON.stringify({
        system_id: systemId,
        ...input,
      }),
    },
  )
}


export async function updateFeedingSchedule(
  scheduleId: string,
  input: FeedingScheduleUpdateInput,
): Promise<FeedingSchedule> {
  return apiRequest<FeedingSchedule>(
    `feeding/schedules/${encodeURIComponent(scheduleId)}`,
    {
      method: 'PATCH',
      body: JSON.stringify(input),
    },
  )
}


export async function deleteFeedingSchedule(
  scheduleId: string,
): Promise<void> {
  await apiRequest<void>(
    `feeding/schedules/${encodeURIComponent(scheduleId)}`,
    {
      method: 'DELETE',
    },
  )
}


export async function loadFeedingEventHistoryPage(
  systemId: string,
  offset: number,
): Promise<FeedingEventHistoryPage> {
  const normalizedOffset =
    Math.max(
      0,
      Math.trunc(offset),
    )

  const encodedSystemId =
    encodeURIComponent(
      systemId,
    )

  const events =
    await apiRequest<FeedingEvent[]>(
      `feeding/events/history?system_id=${encodedSystemId}` +
      `&limit=${FEEDING_HISTORY_PAGE_SIZE + 1}` +
      `&offset=${normalizedOffset}`,
    )

  return {
    events: events.slice(
      0,
      FEEDING_HISTORY_PAGE_SIZE,
    ),

    hasNext:
      events.length >
      FEEDING_HISTORY_PAGE_SIZE,

    hasPrevious:
      normalizedOffset > 0,

    offset: normalizedOffset,
  }
}

import {
  useEffect,
  useState,
} from 'react'

import {
  loadSystemActuatorOverview,
} from '../services/actuatorService'

import {
  listSystems,
} from '../services/dashboardService'

import {
  createFeedingSchedule,
  deleteFeedingSchedule,
  FEEDING_HISTORY_PAGE_SIZE,
  listFeedingSchedules,
  loadFeedingEventHistoryPage,
  updateFeedingSchedule,
} from '../services/feedingService'

import type {
  FeedingEventHistoryPage,
} from '../services/feedingService'

import type {
  AquaponicsSystem,
  DashboardActuator,
} from '../types/dashboard'

import type {
  FeedingSchedule,
  FeedingScheduleInput,
  FeedingScheduleUpdateInput,
} from '../types/feeding'


type LoadStatus =
  'loading' |
  'ready' |
  'error'


type MutationStatus =
  'idle' |
  'saving' |
  'deleting' |
  'error'


interface ActuatorState {
  systemId: string
  status: LoadStatus
  actuators: DashboardActuator[]
}


interface ScheduleState {
  systemId: string
  revision: number
  status: LoadStatus
  schedules: FeedingSchedule[]
}


interface HistoryState {
  systemId: string
  offset: number
  revision: number
  status: LoadStatus
  data: FeedingEventHistoryPage | null
}


interface MutationState {
  systemId: string
  status: MutationStatus
  errorMessage: string | null
}


export function useFeedingOverview() {
  const [
    systems,
    setSystems,
  ] = useState<AquaponicsSystem[]>([])

  const [
    systemsStatus,
    setSystemsStatus,
  ] = useState<LoadStatus>(
    'loading',
  )

  const [
    selection,
    setSelection,
  ] = useState({
    systemId:
      null as string | null,

    offset: 0,
    revision: 0,
  })

  const [
    actuatorState,
    setActuatorState,
  ] = useState<ActuatorState | null>(
    null,
  )

  const [
    scheduleState,
    setScheduleState,
  ] = useState<ScheduleState | null>(
    null,
  )

  const [
    scheduleRevision,
    setScheduleRevision,
  ] = useState(
    0,
  )

  const [
    historyState,
    setHistoryState,
  ] = useState<HistoryState | null>(
    null,
  )

  const [
    mutationState,
    setMutationState,
  ] = useState<MutationState | null>(
    null,
  )


  /*
   * Load systems once.
   */
  useEffect(() => {
    let active = true

    async function load() {
      try {
        const result =
          await listSystems()

        if (!active) {
          return
        }

        const ordered =
          [...result].sort(
            (left, right) =>
              left.name.localeCompare(
                right.name,
              ),
          )

        setSystems(
          ordered,
        )

        setSelection({
          systemId:
            ordered[0]?.id ?? null,

          offset: 0,
          revision: 0,
        })

        setSystemsStatus(
          'ready',
        )
      } catch {
        if (active) {
          setSystemsStatus(
            'error',
          )
        }
      }
    }

    void load()

    return () => {
      active = false
    }
  }, [])


  /*
   * Load all configured, non-deleted actuators
   * belonging to the selected system.
   *
   * Do not guess which actuator types are valid
   * feeders. The backend relationship contract
   * remains authoritative.
   */
  useEffect(() => {
    const systemId =
      selection.systemId

    if (systemId === null) {
      return
    }

    const selectedSystemId =
      systemId

    let active = true

    async function load() {
      try {
        const result =
          await loadSystemActuatorOverview(
            selectedSystemId,
          )

        if (!active) {
          return
        }

        setActuatorState({
          systemId:
            selectedSystemId,

          status: 'ready',

          actuators:
            [...result.actuators].sort(
              (left, right) =>
                left.name.localeCompare(
                  right.name,
                ),
            ),
        })
      } catch {
        if (!active) {
          return
        }

        setActuatorState({
          systemId:
            selectedSystemId,

          status: 'error',
          actuators: [],
        })
      }
    }

    void load()

    return () => {
      active = false
    }
  }, [
    selection.systemId,
  ])


  /*
   * Schedule configuration reloads after a
   * successful create/update/delete.
   */
  useEffect(() => {
    const systemId =
      selection.systemId

    if (systemId === null) {
      return
    }

    const selectedSystemId =
      systemId

    const revision =
      scheduleRevision

    let active = true

    async function load() {
      try {
        const schedules =
          await listFeedingSchedules(
            selectedSystemId,
          )

        if (!active) {
          return
        }

        setScheduleState({
          systemId:
            selectedSystemId,

          revision,
          status: 'ready',
          schedules,
        })
      } catch {
        if (!active) {
          return
        }

        setScheduleState({
          systemId:
            selectedSystemId,

          revision,
          status: 'error',
          schedules: [],
        })
      }
    }

    void load()

    return () => {
      active = false
    }
  }, [
    selection.systemId,
    scheduleRevision,
  ])


  /*
   * Bounded persisted feeding-event history.
   */
  useEffect(() => {
    const {
      systemId,
      offset,
      revision,
    } = selection

    if (systemId === null) {
      return
    }

    const selectedSystemId =
      systemId

    let active = true

    async function load() {
      try {
        const data =
          await loadFeedingEventHistoryPage(
            selectedSystemId,
            offset,
          )

        if (!active) {
          return
        }

        setHistoryState({
          systemId:
            selectedSystemId,

          offset,
          revision,
          status: 'ready',
          data,
        })
      } catch {
        if (!active) {
          return
        }

        setHistoryState({
          systemId:
            selectedSystemId,

          offset,
          revision,
          status: 'error',
          data: null,
        })
      }
    }

    void load()

    return () => {
      active = false
    }
  }, [
    selection,
  ])


  const selectedSystem =
    systems.find(
      (system) =>
        system.id ===
        selection.systemId,
    ) ?? null


  const currentActuatorState =
    actuatorState?.systemId ===
    selection.systemId
      ? actuatorState
      : null


  const currentScheduleState =
    scheduleState?.systemId ===
      selection.systemId &&
    scheduleState?.revision ===
      scheduleRevision
      ? scheduleState
      : null


  const currentHistoryState =
    historyState?.systemId ===
      selection.systemId &&
    historyState?.offset ===
      selection.offset &&
    historyState?.revision ===
      selection.revision
      ? historyState
      : null


  const currentMutationState =
    mutationState?.systemId ===
    selection.systemId
      ? mutationState
      : null


  const actuatorStatus =
    currentActuatorState?.status ??
    'loading'


  const scheduleStatus =
    currentScheduleState?.status ??
    'loading'


  const historyStatus =
    currentHistoryState?.status ??
    'loading'


  const mutationStatus =
    currentMutationState?.status ??
    'idle'


  const hasNext =
    currentHistoryState
      ?.data
      ?.hasNext ?? false


  async function createSchedule(
    input: FeedingScheduleInput,
  ): Promise<FeedingSchedule> {
    const systemId =
      selection.systemId

    if (systemId === null) {
      throw new Error(
        'No aquaponics system selected.',
      )
    }

    setMutationState({
      systemId,
      status: 'saving',
      errorMessage: null,
    })

    try {
      const schedule =
        await createFeedingSchedule(
          systemId,
          input,
        )

      setScheduleRevision(
        (current) =>
          current + 1,
      )

      setMutationState(
        null,
      )

      return schedule
    } catch (error) {
      setMutationState({
        systemId,
        status: 'error',

        errorMessage:
          'Unable to create feeding schedule.',
      })

      throw error
    }
  }


  async function updateSchedule(
    scheduleId: string,
    input: FeedingScheduleUpdateInput,
  ): Promise<FeedingSchedule> {
    const systemId =
      selection.systemId

    if (systemId === null) {
      throw new Error(
        'No aquaponics system selected.',
      )
    }

    setMutationState({
      systemId,
      status: 'saving',
      errorMessage: null,
    })

    try {
      const schedule =
        await updateFeedingSchedule(
          scheduleId,
          input,
        )

      setScheduleRevision(
        (current) =>
          current + 1,
      )

      setMutationState(
        null,
      )

      return schedule
    } catch (error) {
      setMutationState({
        systemId,
        status: 'error',

        errorMessage:
          'Unable to update feeding schedule.',
      })

      throw error
    }
  }


  async function deleteSchedule(
    scheduleId: string,
  ): Promise<void> {
    const systemId =
      selection.systemId

    if (systemId === null) {
      throw new Error(
        'No aquaponics system selected.',
      )
    }

    setMutationState({
      systemId,
      status: 'deleting',
      errorMessage: null,
    })

    try {
      await deleteFeedingSchedule(
        scheduleId,
      )

      setScheduleRevision(
        (current) =>
          current + 1,
      )

      setMutationState(
        null,
      )
    } catch (error) {
      setMutationState({
        systemId,
        status: 'error',

        errorMessage:
          'Unable to delete feeding schedule.',
      })

      throw error
    }
  }


  return {
    systems,
    systemsStatus,

    selectedSystemId:
      selection.systemId,

    selectedSystem,

    actuators:
      currentActuatorState
        ?.actuators ?? [],

    actuatorStatus,

    schedules:
      currentScheduleState
        ?.schedules ?? [],

    scheduleStatus,

    events:
      currentHistoryState
        ?.data
        ?.events ?? [],

    historyStatus,

    mutationStatus,

    mutationError:
      currentMutationState
        ?.errorMessage ?? null,

    isMutating:
      mutationStatus ===
        'saving' ||
      mutationStatus ===
        'deleting',

    hasNext,

    hasPrevious:
      selection.offset > 0,

    pageNumber:
      Math.floor(
        selection.offset /
          FEEDING_HISTORY_PAGE_SIZE,
      ) + 1,

    createSchedule,
    updateSchedule,
    deleteSchedule,


    selectSystem(
      systemId: string,
    ) {
      if (
        !systems.some(
          (system) =>
            system.id === systemId,
        )
      ) {
        return
      }

      setMutationState(
        null,
      )

      setSelection(
        (current) => {
          if (
            current.systemId ===
            systemId
          ) {
            return current
          }

          return {
            systemId,
            offset: 0,

            revision:
              current.revision + 1,
          }
        },
      )
    },


    nextPage() {
      if (
        historyStatus !== 'ready' ||
        !hasNext
      ) {
        return
      }

      setSelection(
        (current) => ({
          ...current,

          offset:
            current.offset +
            FEEDING_HISTORY_PAGE_SIZE,

          revision:
            current.revision + 1,
        }),
      )
    },


    previousPage() {
      if (
        historyStatus ===
          'loading' ||
        selection.offset === 0
      ) {
        return
      }

      setSelection(
        (current) => ({
          ...current,

          offset:
            Math.max(
              0,
              current.offset -
                FEEDING_HISTORY_PAGE_SIZE,
            ),

          revision:
            current.revision + 1,
        }),
      )
    },
  }
}

import {
  useState,
} from 'react'

import {
  useAuth,
} from '../hooks/useAuth'

import {
  useFeedingOverview,
} from '../hooks/useFeedingOverview'

import type {
  FeedingSchedule,
  FeedingScheduleInput,
  FeedingScheduleUpdateInput,
} from '../types/feeding'


interface ScheduleDraft {
  actuatorId: string
  name: string
  scheduleExpression: string
  enabled: boolean
  maxPerFeedGrams: string
}


interface ScheduleEditor {
  systemId: string
  mode: 'create' | 'edit'
  scheduleId: string | null
  original: FeedingSchedule | null
  draft: ScheduleDraft
}


function formatTimestamp(
  value: string,
): string {
  const date =
    new Date(value)

  return Number.isNaN(
    date.getTime(),
  )
    ? value
    : date.toLocaleString()
}


function formatNullableValue(
  value:
    number |
    string |
    null,
): string {
  return value === null
    ? 'Not recorded'
    : String(value)
}


export function FeedingPage() {
  const overview =
    useFeedingOverview()

  const auth =
    useAuth()

  const [
    editor,
    setEditor,
  ] = useState<ScheduleEditor | null>(
    null,
  )

  const [
    formError,
    setFormError,
  ] = useState<string | null>(
    null,
  )


  const canManageSchedules =
    auth.user?.role === 'admin'


  const currentEditor =
    editor?.systemId ===
    overview.selectedSystemId
      ? editor
      : null


  const enabledSchedules =
    overview.schedules.filter(
      (schedule) =>
        schedule.enabled,
    ).length


  function openCreateEditor() {
    const firstActuator =
      overview.actuators[0]

    if (
      overview.selectedSystemId ===
        null ||
      firstActuator === undefined
    ) {
      return
    }

    setFormError(
      null,
    )

    setEditor({
      systemId:
        overview.selectedSystemId,

      mode: 'create',
      scheduleId: null,
      original: null,

      draft: {
        actuatorId:
          firstActuator.id,

        name: '',

        scheduleExpression:
          '',

        enabled: true,

        maxPerFeedGrams:
          '',
      },
    })
  }


  function openEditEditor(
    schedule: FeedingSchedule,
  ) {
    if (
      overview.selectedSystemId ===
      null
    ) {
      return
    }

    setFormError(
      null,
    )

    setEditor({
      systemId:
        overview.selectedSystemId,

      mode: 'edit',

      scheduleId:
        schedule.id,

      original:
        schedule,

      draft: {
        actuatorId:
          schedule.actuator_id,

        name:
          schedule.name,

        scheduleExpression:
          schedule
            .schedule_expression,

        enabled:
          schedule.enabled,

        maxPerFeedGrams:
          String(
            schedule
              .max_per_feed_grams,
          ),
      },
    })
  }


  function updateDraft(
    update:
      Partial<ScheduleDraft>,
  ) {
    setEditor(
      (current) => {
        if (current === null) {
          return current
        }

        return {
          ...current,

          draft: {
            ...current.draft,
            ...update,
          },
        }
      },
    )
  }


  async function submitEditor() {
    if (currentEditor === null) {
      return
    }

    const name =
      currentEditor
        .draft
        .name
        .trim()

    const scheduleExpression =
      currentEditor
        .draft
        .scheduleExpression
        .trim()

    const maxPerFeedGrams =
      Number(
        currentEditor
          .draft
          .maxPerFeedGrams,
      )


    if (name.length === 0) {
      setFormError(
        'Schedule name is required.',
      )

      return
    }


    if (
      scheduleExpression.length ===
      0
    ) {
      setFormError(
        'Schedule expression is required.',
      )

      return
    }


    if (
      currentEditor
        .draft
        .actuatorId
        .length === 0
    ) {
      setFormError(
        'An actuator is required.',
      )

      return
    }


    if (
      !Number.isFinite(
        maxPerFeedGrams,
      ) ||
      maxPerFeedGrams <= 0
    ) {
      setFormError(
        'Max per feed must be greater than zero.',
      )

      return
    }


    setFormError(
      null,
    )


    try {
      if (
        currentEditor.mode ===
        'create'
      ) {
        const input:
          FeedingScheduleInput = {
            actuator_id:
              currentEditor
                .draft
                .actuatorId,

            name,

            schedule_expression:
              scheduleExpression,

            enabled:
              currentEditor
                .draft
                .enabled,

            max_per_feed_grams:
              maxPerFeedGrams,
          }

        await overview.createSchedule(
          input,
        )

        setEditor(
          null,
        )

        return
      }


      const original =
        currentEditor.original

      const scheduleId =
        currentEditor.scheduleId

      if (
        original === null ||
        scheduleId === null
      ) {
        return
      }


      const update:
        FeedingScheduleUpdateInput =
        {}


      if (
        currentEditor
          .draft
          .actuatorId !==
        original.actuator_id
      ) {
        update.actuator_id =
          currentEditor
            .draft
            .actuatorId
      }


      if (
        name !==
        original.name
      ) {
        update.name =
          name
      }


      if (
        scheduleExpression !==
        original
          .schedule_expression
      ) {
        update.schedule_expression =
          scheduleExpression
      }


      if (
        currentEditor
          .draft
          .enabled !==
        original.enabled
      ) {
        update.enabled =
          currentEditor
            .draft
            .enabled
      }


      if (
        maxPerFeedGrams !==
        original
          .max_per_feed_grams
      ) {
        update.max_per_feed_grams =
          maxPerFeedGrams
      }


      if (
        Object.keys(
          update,
        ).length === 0
      ) {
        setEditor(
          null,
        )

        return
      }


      await overview.updateSchedule(
        scheduleId,
        update,
      )

      setEditor(
        null,
      )
    } catch {
      /*
       * The hook owns the authoritative
       * mutation error state.
       */
    }
  }

                      .name
                  }
                </p>
              </div>


              {overview.scheduleStatus ===
                'ready' && (
                <span
                  className={
                    'sensor-detail-history-page'
                  }
                >
                  {
                    overview
                      .schedules
                      .length
                  } configured ·{' '}
                  {
                    enabledSchedules
                  } enabled
                </span>
              )}
            </div>


            <p
              className={
                'actuator-hardware-note'
              }
            >
              Feeding schedules are persisted
              configuration. Creating or changing
              a schedule does not itself confirm
              physical feeder execution.
            </p>


            {canManageSchedules && (
              <div>
                <button
                  type="button"
                  className={
                    'sensor-history-button'
                  }
                  disabled={
                    overview.isMutating ||
                    overview.actuatorStatus !==
                      'ready' ||
                    overview.actuators.length ===
                      0
                  }
                  onClick={
                    openCreateEditor
                  }
                >
                  Add schedule
                </button>

                {overview.actuatorStatus ===
                  'ready' &&
                  overview.actuators.length ===
                    0 && (
                  <p>
                    No configured actuator is
                    available for this system.
                  </p>
                )}
              </div>
            )}


            {overview.mutationError !==
              null && (
              <div
                role="alert"
                className={
                  'dashboard-message ' +
                  'dashboard-message--error'
                }
              >
                {overview.mutationError}
              </div>
            )}


            {currentEditor !== null && (
              <form
                onSubmit={
                  (event) => {
                    event.preventDefault()

                    void submitEditor()
                  }
                }
              >
                <h3>
                  {currentEditor.mode ===
                  'create'
                    ? 'Add feeding schedule'
                    : 'Edit feeding schedule'}
                </h3>


                <div
                  className="system-selector"
                >
                  <label
                    htmlFor={
                      'feeding-schedule-name'
                    }
                  >
                    Name
                  </label>

                  <input
                    id={
                      'feeding-schedule-name'
                    }
                    value={
                      currentEditor
                        .draft
                        .name
                    }
                    disabled={
                      overview.isMutating
                    }
                    onChange={
                      (event) =>
                        updateDraft({
                          name:
                            event
                              .target
                              .value,
                        })
                    }
                  />
                </div>


                <div
                  className="system-selector"
                >
                  <label
                    htmlFor={
                      'feeding-schedule-actuator'
                    }
                  >
                    Actuator
                  </label>

                  <select
                    id={
                      'feeding-schedule-actuator'
                    }
                    value={
                      currentEditor
                        .draft
                        .actuatorId
                    }
                    disabled={
                      overview.isMutating
                    }
                    onChange={
                      (event) =>
                        updateDraft({
                          actuatorId:
                            event
                              .target
                              .value,
                        })
                    }
                  >
                    {!overview.actuators.some(
                      (actuator) =>
                        actuator.id ===
                        currentEditor
                          .draft
                          .actuatorId,
                    ) && (
                      <option
                        value={
                          currentEditor
                            .draft
                            .actuatorId
                        }
                      >
                        Current actuator
                        unavailable —{' '}
                        {
                          currentEditor
                            .draft
                            .actuatorId
                        }
                      </option>
                    )}

                    {overview.actuators.map(
                      (actuator) => (
                        <option
                          key={
                            actuator.id
                          }
                          value={
                            actuator.id
                          }
                        >
                          {actuator.name}
                          {' — '}
                          {
                            actuator
                              .actuator_type
                          }
                        </option>
                      ),
                    )}
                  </select>
                </div>


                <div
                  className="system-selector"
                >
                  <label
                    htmlFor={
                      'feeding-schedule-expression'
                    }
                  >
                    Schedule expression
                  </label>

                  <input
                    id={
                      'feeding-schedule-expression'
                    }
                    value={
                      currentEditor
                        .draft
                        .scheduleExpression
                    }
                    disabled={
                      overview.isMutating
                    }
                    onChange={
                      (event) =>
                        updateDraft({
                          scheduleExpression:
                            event
                              .target
                              .value,
                        })
                    }
                  />

                  <small>
                    Stored exactly as entered.
                    This interface does not
                    interpret the expression.
                  </small>
                </div>


                <div
                  className="system-selector"
                >
                  <label
                    htmlFor={
                      'feeding-max-per-feed'
                    }
                  >
                    Max per feed (grams)
                  </label>

                  <input
                    id={
                      'feeding-max-per-feed'
                    }
                    type="number"
                    min="0"
                    step="any"
                    value={
                      currentEditor
                        .draft
                        .maxPerFeedGrams
                    }
                    disabled={
                      overview.isMutating
                    }
                    onChange={
                      (event) =>
                        updateDraft({
                          maxPerFeedGrams:
                            event
                              .target
                              .value,
                        })
                    }
                  />
                </div>


                <label>
                  <input
                    type="checkbox"
                    checked={
                      currentEditor
                        .draft
                        .enabled
                    }
                    disabled={
                      overview.isMutating
                    }
                    onChange={
                      (event) =>
                        updateDraft({
                          enabled:
                            event
                              .target
                              .checked,
                        })
                    }
                  />

                  {' '}Enabled
                </label>


                {formError !== null && (
                  <div
                    role="alert"
                    className={
                      'dashboard-message ' +
                      'dashboard-message--error'
                    }
                  >
                    {formError}
                  </div>
                )}


                <div>
                  <button
                    type="submit"
                    className={
                      'sensor-history-button'
                    }
                    disabled={
                      overview.isMutating
                    }
                  >
                    {overview.mutationStatus ===
                    'saving'
                      ? 'Saving...'
                      : 'Save schedule'}
                  </button>

                  <button
                    type="button"
                    className={
                      'sensor-history-button'
                    }
                    disabled={
                      overview.isMutating
                    }
                    onClick={
                      () => {
                        setEditor(
                          null,
                        )

                        setFormError(
                          null,
                        )
                      }
                    }
                  >
                    Cancel
                  </button>
                </div>
              </form>
            )}


            {overview.scheduleStatus ===
              'loading' && (
              <div role="status">
                Loading feeding schedules...
              </div>
            )}


            {overview.scheduleStatus ===
              'error' && (
              <div
                role="alert"
                className={
                  'dashboard-message ' +
                  'dashboard-message--error'
                }
              >
                Unable to load feeding schedules
                for the selected system.
              </div>
            )}


            {overview.scheduleStatus ===
              'ready' &&
              (
                overview.schedules.length ===
                0
                  ? (
                    <p>
                      No feeding schedules
                      configured for this
                      system.
                    </p>
                  )
                  : (
                    <div
                      className={
                        'sensor-history-table-wrap'
                      }
                    >
                      <table
                        className={
                          'sensor-history-table'
                        }
                        aria-label={
                          'Configured feeding schedules'
                        }
                      >
                        <thead>
                          <tr>
                            <th>Name</th>
                            <th>Status</th>
                            <th>
                              Schedule expression
                            </th>
                            <th>
                              Max per feed
                              (grams)
                            </th>
                            <th>
                              Actuator ID
                            </th>

                            {canManageSchedules && (
                              <th>
                                Actions
                              </th>
                            )}
                          </tr>
                        </thead>

                        <tbody>
                          {
                            overview
                              .schedules
                              .map(
                                (schedule) => (
                                  <tr
                                    key={
                                      schedule.id
                                    }
                                  >
                                    <td>
                                      {
                                        schedule
                                          .name
                                      }
                                    </td>

                                    <td>
                                      {
                                        schedule
                                          .enabled
                                          ? 'Enabled'
                                          : 'Disabled'
                                      }
                                    </td>

                                    <td>
                                      {
                                        schedule
                                          .schedule_expression
                                      }
                                    </td>

                                    <td>
                                      {
                                        schedule
                                          .max_per_feed_grams
                                      }
                                    </td>

                                    <td>
                                      {
                                        schedule
                                          .actuator_id
                                      }
                                    </td>


                                    {canManageSchedules && (
                                      <td>
                                        <button
                                          type="button"
                                          className={
                                            'sensor-history-button'
                                          }
                                          disabled={
                                            overview
                                              .isMutating
                                          }
                                          onClick={
                                            () =>
                                              openEditEditor(
                                                schedule,
                                              )
                                          }
                                        >
                                          Edit
                                        </button>

                                        <button
                                          type="button"
                                          className={
                                            'sensor-history-button'
                                          }
                                          disabled={
                                            overview
                                              .isMutating
                                          }
                                          onClick={
                                            () => {
                                              void removeSchedule(
                                                schedule,
                                              )
                                            }
                                          }
                                        >
                                          Delete
                                        </button>
                                      </td>
                                    )}
                                  </tr>
                                ),
                              )
                          }
                        </tbody>
                      </table>
                    </div>
                  )
              )}
          </section>


          <section
            className="sensor-detail-panel"
            aria-labelledby={
              'feeding-history-heading'
            }
          >
            <div
              className={
                'sensor-detail-history-heading'
              }
            >
              <div>
                <h2
                  id={
                    'feeding-history-heading'
                  }
                >
                  Persisted feeding event
                  history
                </h2>

                <p>
                  {
                    overview
                      .selectedSystem
                      .name
                  }
                </p>
              </div>

              <span
                className={
                  'sensor-detail-history-page'
                }
              >
                Page {
                  overview.pageNumber
                }
              </span>
            </div>


            <p
              className={
                'actuator-hardware-note'
              }
            >
              Feeding events are persisted
              application records and do not by
              themselves confirm physical feeder
              execution.
            </p>


            {overview.historyStatus ===
              'loading' && (
              <div role="status">
                Loading feeding event
                history...
              </div>
            )}


            {overview.historyStatus ===
              'error' && (
              <div
                role="alert"
                className={
                  'dashboard-message ' +
                  'dashboard-message--error'
                }
              >
                Unable to load feeding event
                history for the selected
                system.
              </div>
            )}


            {overview.historyStatus ===
              'ready' &&
              (
                overview.events.length ===
                0
                  ? (
                    <p>
                      No persisted feeding
                      events on this page.
                    </p>
                  )
                  : (
                    <div
                      className={
                        'sensor-history-table-wrap'
                      }
                    >
                      <table
                        className={
                          'sensor-history-table'
                        }
                        aria-label={
                          'Persisted feeding event history'
                        }
                      >
                        <thead>
                          <tr>
                            <th>Status</th>
                            <th>Executed</th>
                            <th>
                              Amount (grams)
                            </th>
                            <th>
                              Duration (seconds)
                            </th>
                            <th>Tank ID</th>
                            <th>Schedule ID</th>
                            <th>
                              Actuator event ID
                            </th>
                            <th>
                              Biomass basis
                              (grams)
                            </th>
                            <th>
                              Calibration ID
                            </th>
                            <th>Notes</th>
                          </tr>
                        </thead>

                        <tbody>
                          {
                            overview
                              .events
                              .map(
                                (event) => (
                                  <tr
                                    key={
                                      event.id
                                    }
                                  >
                                    <td>
                                      {
                                        event.status
                                      }
                                    </td>

                                    <td>
                                      {
                                        formatTimestamp(
                                          event
                                            .executed_at,
                                        )
                                      }
                                    </td>

                                    <td>
                                      {
                                        event
                                          .amount_grams
                                      }
                                    </td>

                                    <td>
                                      {
                                        event
                                          .duration_seconds
                                      }
                                    </td>

                                    <td>
                                      {
                                        event
                                          .tank_id
                                      }
                                    </td>

                                    <td>
                                      {
                                        formatNullableValue(
                                          event
                                            .schedule_id,
                                        )
                                      }
                                    </td>

                                    <td>
                                      {
                                        event
                                          .actuator_event_id
                                      }
                                    </td>

                                    <td>
                                      {
                                        formatNullableValue(
                                          event
                                            .calculated_from_biomass,
                                        )
                                      }
                                    </td>

                                    <td>
                                      {
                                        formatNullableValue(
                                          event
                                            .calibration_id,
                                        )
                                      }
                                    </td>

                                    <td>
                                      {
                                        formatNullableValue(
                                          event
                                            .notes,
                                        )
                                      }
                                    </td>
                                  </tr>
                                ),
                              )
                          }
                        </tbody>
                      </table>
                    </div>
                  )
              )}


            <footer
              className={
                'sensor-history-pagination'
              }
            >
              <span>
                Up to 50 feeding events
                per page
              </span>

              <div>
                <button
                  type="button"
                  className={
                    'sensor-history-button'
                  }
                  disabled={
                    !overview.hasPrevious ||
                    overview.historyStatus ===
                      'loading'
                  }
                  onClick={
                    overview.previousPage
                  }
                >
                  Previous
                </button>

                <button
                  type="button"
                  className={
                    'sensor-history-button'
                  }
                  disabled={
                    !overview.hasNext ||
                    overview.historyStatus !==
                      'ready'
                  }
                  onClick={
                    overview.nextPage
                  }
                >
                  Next
                </button>
              </div>
            </footer>
          </section>
        </>
      )}
    </section>
  )
}
