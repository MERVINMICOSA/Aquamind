@router.get(
    "/population",
    response_model=list[FishPopulationRecordResponse],
)
async def list_population_records(
    tank_id: UUID | None = None,
    session: AsyncSession = Depends(get_db_session),
):
    query = (
        select(FishPopulationRecord)
        .order_by(FishPopulationRecord.effective_at.desc())
    )

    if tank_id is not None:
        query = query.where(
            FishPopulationRecord.tank_id == tank_id
        )

    result = await session.execute(query)

    return result.scalars().all()


@router.get(
    "/population/{record_id}",
    response_model=FishPopulationRecordResponse,
)
async def get_population_record(
    record_id: UUID,
    session: AsyncSession = Depends(get_db_session),
):
    record = await session.get(
        FishPopulationRecord,
        record_id,
    )

    if record is None:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Fish population record not found",
        )

    return record


@router.post(
    "/population",
    response_model=FishPopulationRecordResponse,
    status_code=status.HTTP_201_CREATED,
)
async def create_population_record(
    payload: FishPopulationRecordCreate,
    session: AsyncSession = Depends(
        get_db_session
    ),
    current_user: User = Depends(
        get_current_user
    ),
):
    tank = await session.get(
        Tank,
        payload.tank_id,
    )

    if (
        tank is None
        or tank.deleted_at is not None
    ):
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Tank not found",
        )

    record = FishPopulationRecord(
        tank_id=payload.tank_id,
        species=payload.species,
        count=payload.count,
        estimated_biomass_grams=(
            payload.estimated_biomass_grams
        ),
        effective_at=payload.effective_at,

        # Recorder identity is authoritative from
        # authentication, never from the request.
        recorded_by_user_id=current_user.id,

        notes=payload.notes,
    )

    session.add(
        record
    )

    await session.commit()
    await session.refresh(
        record
    )

    await create_audit_log(
        session,
        user_id=current_user.id,
        action=AuditAction.CREATE,
        entity_type="fish_population_record",
        entity_id=record.id,
        new_value=(
            _fish_population_record_audit_snapshot(
                record
            )
        ),
    )

    return record


@router.patch(
    "/population/{record_id}",
    response_model=FishPopulationRecordResponse,
)
async def update_population_record(
    record_id: UUID,
    payload: FishPopulationRecordUpdate,
    session: AsyncSession = Depends(
        get_db_session
    ),
    current_user: User = Depends(
        get_current_user
    ),
):
    record = await session.get(
        FishPopulationRecord,
        record_id,
    )

    if record is None:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Fish population record not found",
        )

    updates = payload.model_dump(
        exclude_unset=True
    )

    # Preserve original recorder provenance.
    # The authenticated modifier is instead captured
    # by AuditLog.user_id.
    updates.pop(
        "recorded_by_user_id",
        None,
    )

    old_value = {
        field: _audit_value(
            getattr(
                record,
                field,
            )
        )
        for field in updates
    }

    for field, value in updates.items():
        setattr(
            record,
            field,
            value,
        )

    await session.commit()
    await session.refresh(
        record
    )

    new_value = {
        field: _audit_value(
            getattr(
                record,
                field,
            )
        )
        for field in updates
    }

    await create_audit_log(
        session,
        user_id=current_user.id,
        action=AuditAction.UPDATE,
        entity_type="fish_population_record",
        entity_id=record.id,
        old_value=old_value,
        new_value=new_value,
    )

    return record


@router.delete(
    "/population/{record_id}",
    status_code=status.HTTP_204_NO_CONTENT,
)
async def delete_population_record(
    record_id: UUID,
    session: AsyncSession = Depends(
        get_db_session
    ),
    current_user: User = Depends(
        require_admin
    ),
):
    record = await session.get(
        FishPopulationRecord,
        record_id,
    )

    if record is None:
        raise HTTPException(
            status_code=status.HTTP_404_NOT_FOUND,
            detail="Fish population record not found",
        )

    old_value = (
        _fish_population_record_audit_snapshot(
            record
        )
    )

    entity_id = record.id

    await session.delete(
        record
    )

    await session.commit()

    await create_audit_log(
        session,
        user_id=current_user.id,
        action=AuditAction.DELETE,
        entity_type="fish_population_record",
        entity_id=entity_id,
        old_value=old_value,
        new_value=None,
    )
